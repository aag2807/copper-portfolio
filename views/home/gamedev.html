function love.load()
    world = love.physics.newWorld(0, 9.81*64)
    player = entities.spawn("player", 100, 100)
end

function love.update(dt)
    accumulator = accumulator + dt
    while accumulator >= STEP do
        world:update(STEP)
        input.poll()
        ai.tick(STEP)
        accumulator = accumulator - STEP
    end
end

function love.draw()
    camera:attach()
    tilemap:draw()
    entities:draw()
    camera:detach()
    hud:draw()
end
